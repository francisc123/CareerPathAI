\chapter{Executive Summary}
% ============================================================

\begin{tcolorbox}[
  colback=primary!6!white, colframe=primary, arc=6pt,
  title={\faInfoCircle\quad Platform Overview},
  fonttitle=\bfseries\color{white},
  coltitle=white, colbacktitle=primary
]
CareerPath AI is an intelligent, event-driven career development platform that combines
large language models, semantic vector search, and automated job matching into a single,
cohesive product. The system guides users from CV submission to job placement through a
fully automated AI pipeline.
\end{tcolorbox}

\vspace{0.5cm}

\section{Key Value Propositions}

\begin{itemize}[leftmargin=*, label=\textcolor{primary}{\faCheckCircle}]
  \item \textbf{Automated CV Analysis} — Gemini 2.5 Flash scores CVs across five weighted dimensions and identifies skill gaps
  \item \textbf{Semantic Job Matching} — Qdrant vector similarity search over 3072-dimensional embeddings (gemini-embedding-001)
  \item \textbf{Personalized Learning Paths} — Dynamic module generation via Udemy/YouTube APIs based on gap analysis
  \item \textbf{AI Interview Simulation} — Real-time HR and Technical interviews via WebSocket with streaming Gemini responses
  \item \textbf{Shadow Profiling} — Agent Orchestrator silently builds user competency profiles from natural conversation
\end{itemize}

\section{Embedding Pipeline}

\begin{tcolorbox}[colback=darkgray!8, colframe=primary!40, arc=5pt]
\small
\begin{enumerate}[leftmargin=*]
  \item PDF text extracted via \textbf{pdfplumber} in cv-service
  \item Text chunked and embedded with \textbf{gemini-embedding-001} (dim = 3072)
  \item Vector stored in \textbf{Qdrant} collection \texttt{cv\_embeddings} (Cosine similarity)
  \item Job descriptions embedded identically into \texttt{job\_embeddings}
  \item Semantic nearest-neighbor search retrieves top-$k$ job matches
\end{enumerate}
\end{tcolorbox}

\chapter{Glossary}
% ============================================================

\begin{description}[leftmargin=3cm, style=nextline]
  \item[RAG] Retrieval-Augmented Generation — combining vector search with LLM generation
  \item[Qdrant] Open-source vector database used for semantic similarity search
  \item[Redpanda] Kafka-compatible event streaming platform with lower operational overhead
  \item[LangGraph] Framework for building stateful, multi-agent AI workflows
  \item[Gemini] Google's multimodal large language model family (using 2.5 Flash)
  \item[gemini-embedding-001] Google's text embedding model producing 3072-dimensional vectors
  \item[RS256] RSA-based JWT signing algorithm using asymmetric key pairs
  \item[Keycloak] Open-source Identity and Access Management solution with OAuth2/OIDC
  \item[Shadow Profiling] Silent user profiling extracted from natural conversation by the orchestrator
  \item[aiokafka] Async Python Kafka client used by all FastAPI agents
  \item[MinIO] S3-compatible object storage used for raw CV PDF files
  \item[SSE] Server-Sent Events — used by notification-service for real-time push notifications
\end{description}
